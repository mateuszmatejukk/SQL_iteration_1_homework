/*Marketing porownuje kanaly pozyskania klientow z Polski i Niemiec. 
 * Chce zobaczyc tylko kanaly, z ktorych pochodza przynajmniej dwa oplacone zamowienia.

Wynik: acquisition_channel, customers_count, paying_customers_count, paid_orders_count, paid_revenue.

customers_count ma uwzgledniac wszystkich klientow danego kanalu z tych krajow, takze bez oplaconych zamowien. 
paying_customers_count to liczba unikalnych klientow z przynajmniej jednym oplaconym zamowieniem. 
Brak kanalu pokaz jako unknown. 
Posortuj po przychodzie malejaco, potem po kanale.*/
	select
	c.acquisition_channel,
	count(distinct c.customer_id) as customers_count,
	count(distinct o.customer_id) as paying_customers_count,
	count(distinct o.order_id),
	sum(o.total_amount) as paid_revenue,
	coalesce(c.acquisition_channel, 'unknown') as acquisition_channel 
	from course.customers c 
	left join course.orders o
	on c.customer_id = o.customer_id and
	o.status = 'paid' 
	group by c.acquisition_channel
	having count(o.order_id) >=2 
	order by sum(o.total_amount),c.acquisition_channel
